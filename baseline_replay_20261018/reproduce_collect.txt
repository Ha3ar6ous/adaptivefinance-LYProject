# Compare numeric values recursively; configuration hash strings are provenance, not numerical headlines.
def numeric_diff(a,b,path=''):
 out=[]
 if isinstance(a,dict) and isinstance(b,dict):
  for k in sorted(set(a)&set(b)):
   if any(t in k.lower() for t in ['latency','time_ms','wall_seconds','seconds','timestamp']):continue
   out.extend(numeric_diff(a[k],b[k],path+'/'+str(k)))
 elif isinstance(a,list) and isinstance(b,list):
  if len(a)!=len(b):out.append(dict(path=path+'/length',before=len(a),after=len(b)))
  for i,(x,y) in enumerate(zip(a,b)):out.extend(numeric_diff(x,y,path+'/'+str(i)))
 elif isinstance(a,(int,float)) and not isinstance(a,bool) and isinstance(b,(int,float)) and not isinstance(b,bool) and a!=b:
  out.append(dict(path=path,before=a,after=b,absolute_change=b-a,same_at_six_decimals=round(a,6)==round(b,6)))
 elif isinstance(a,bool) and isinstance(b,bool) and a!=b:out.append(dict(path=path,before=a,after=b))
 return out
